        // Hàm hỗ trợ kiểm tra xem 1 công việc có bị QUÁ HẠN CHÓT hay không
        function isTaskOverdue(dateStr, status) {
            if (status === 'Hoàn thành' || !dateStr) return false;

            const parts = dateStr.split('/');
            if (parts.length !== 3) return false;

            const day = parseInt(parts[0], 10);
            const month = parseInt(parts[1], 10) - 1; 
            const year = parseInt(parts[2], 10);

            const dueDate = new Date(year, month, day, 23, 59, 59);
            const now = new Date();

            return dueDate < now;
        }

        // Thêm công việc mới
        function addNewTask() {
            const taskName = prompt("Nhập tên công việc mới:");
            if (!taskName || !taskName.trim()) return;

            const priority = prompt("Mức độ ưu tiên (Cao / Bình thường / Thấp):", "Bình thường") || "Bình thường";
            const date = prompt("Hạn chót (ngày/tháng/năm, VD: 30/09/2026):", "30/09/2026") || "30/09/2026";

            const newTask = {
                id: 'T' + String(tasks.length + 1).padStart(3, '0'),
                name: taskName.trim(),
                user: targetUser,
                status: 'Chưa làm',
                date: date.trim(),
                priority: ['Cao', 'Bình thường', 'Thấp'].includes(priority) ? priority : 'Bình thường'
            };

            tasks.push(newTask);
            saveUserData();
            initApp();
        }

        // Chỉnh sửa công việc
        function editTask(index) {
            const task = tasks[index];

            const newName = prompt("Sửa tên công việc:", task.name);
            if (newName === null) return;

            const newPriority = prompt("Cập nhật mức ưu tiên (Cao / Bình thường / Thấp):", task.priority);
            const newStatus = prompt("Cập nhật trạng thái (Chưa làm / Đang làm / Hoàn thành):", task.status);
            const newDate = prompt("Cập nhật hạn chót (ngày/tháng/năm, VD: 30/09/2026):", task.date);

            if (newName.trim() !== '') task.name = newName.trim();
            if (['Cao', 'Bình thường', 'Thấp'].includes(newPriority)) task.priority = newPriority;
            if (['Chưa làm', 'Đang làm', 'Hoàn thành'].includes(newStatus)) task.status = newStatus;
            if (newDate && newDate.trim() !== '') task.date = newDate.trim();

            saveUserData();
            initApp();
        }

        // Xóa công việc
        function deleteTask(index) {
            if (confirm("Bạn có chắc chắn muốn xóa công việc này?")) {
                tasks.splice(index, 1);
                saveUserData();
                initApp();
            }
        }

        // Render Bảng Kanban
        function renderKanban() {
            const todoBox = document.getElementById('cards-todo');
            const doingBox = document.getElementById('cards-doing');
            const doneBox = document.getElementById('cards-done');

            todoBox.innerHTML = ''; doingBox.innerHTML = ''; doneBox.innerHTML = '';

            let countTodo = 0, countDoing = 0, countDone = 0;

            tasks.forEach(task => {
                const overdue = isTaskOverdue(task.date, task.status);
                const cardOverdueClass = overdue ? 'card-overdue' : '';
                const overdueTag = overdue ? `<span style="color: #dc2626; font-weight: bold; font-size: 11px;"><i class="fa-solid fa-triangle-exclamation"></i> Quá hạn</span>` : '';

                const cardHTML = `
                    <div class="kanban-card ${cardOverdueClass}" draggable="true" ondragstart="drag(event, '${task.id}')">
                        <div style="display: flex; justify-content: space-between; align-items: center;">
                            <span class="id">${task.id}</span>
                            ${overdueTag}
                        </div>
                        <div class="title">${task.name}</div>
                        <div class="meta">
                            <span><i class="fa-regular fa-user"></i> ${task.user}</span>
                            <span style="${overdue ? 'color: #dc2626; font-weight: bold;' : ''}"><i class="fa-regular fa-calendar"></i> ${task.date}</span>
                        </div>
                    </div>
                `;

                if (task.status === 'Chưa làm') {
                    todoBox.innerHTML += cardHTML;
                    countTodo++;
                } else if (task.status === 'Đang làm') {
                    doingBox.innerHTML += cardHTML;
                    countDoing++;
                } else if (task.status === 'Hoàn thành') {
                    doneBox.innerHTML += cardHTML;
                    countDone++;
                }
            });

            document.getElementById('count-todo').innerText = countTodo;
            document.getElementById('count-doing').innerText = countDoing;
            document.getElementById('count-done').innerText = countDone;
        }

            kpiList.push({
                title: title.trim(),
                maxScore: maxScore,
                selfScore: maxScore,
                adminScore: adminMaxScore,
                adminMaxScore: adminMaxScore

            saveKPIRatingStorage();
            renderKPITable();
        }

        // Admin Sửa tên tiêu chí và điểm tối đa
        function editKPICriteria(index) {
            if (!isAdmin) return;
            const item = kpiList[index];
            const newTitle = prompt("Sửa tên tiêu chí:", item.title);
            if (newTitle === null) return;

            const newMaxScoreInput = prompt("Sửa Điểm tối đa TỰ CHẤM:", item.maxScore);
            const newMaxScore = Number(newMaxScoreInput) > 0 ? Number(newMaxScoreInput) : item.maxScore;

            const newAdminMaxScoreInput = prompt("Sửa Điểm tối đa LÃNH ĐẠO CHẤM:", item.adminMaxScore);
            const newAdminMaxScore = Number(newAdminMaxScoreInput) > 0 ? Number(newAdminMaxScoreInput) : item.adminMaxScore;

            if (newTitle.trim() !== '') item.title = newTitle.trim();
            item.maxScore = newMaxScore;
            item.adminMaxScore = newAdminMaxScore;

            if (item.selfScore > newMaxScore) item.selfScore = newMaxScore;
            if (item.adminScore > newAdminMaxScore) item.adminScore = newAdminMaxScore;

            saveKPIRatingStorage();
            renderKPITable();
        }

        // Admin Xóa tiêu chí
        function deleteKPICriteria(index) {
            if (!isAdmin) return;
            if (confirm("Bạn có chắc chắn muốn xóa tiêu chí đánh giá này?")) {
                kpiList.splice(index, 1);
                saveKPIRatingStorage();
                renderKPITable();
            }
        }

        // Cập nhật điểm TỰ CHẤM
        function updateKPISelfScore(index, value) {
            let val = Number(value);
            if (isNaN(val)) val = 0;
            if (val < 0) val = 0;
            if (val > kpiList[index].maxScore) val = kpiList[index].maxScore;

            kpiList[index].selfScore = val;
            calcKPITotal();
        }

        // Cập nhật điểm LÃNH ĐẠO CHẤM
        function updateKPIAdminScore(index, value) {
            if (!isAdmin) return;
            let val = Number(value);
            if (isNaN(val)) val = 0;
            if (val < 0) val = 0;
            if (val > kpiList[index].adminMaxScore) val = kpiList[index].adminMaxScore;

            kpiList[index].adminScore = val;
            calcKPITotal();
        }

        // Tính tổng điểm KPI cho cả 2 luồng
        function calcKPITotal() {
            let selfTotal = 0;
            let maxTotal = 0;
            let adminTotal = 0;
            let adminMaxTotal = 0;

            kpiList.forEach(item => {
                selfTotal += Number(item.selfScore || 0);
                maxTotal += Number(item.maxScore || 0);
                adminTotal += Number(item.adminScore || 0);
                adminMaxTotal += Number(item.adminMaxScore || 0);
            });

            document.getElementById('kpi-total-self').innerText = selfTotal.toFixed(1);
            document.getElementById('kpi-total-max').innerText = maxTotal;

            document.getElementById('kpi-total-admin').innerText = adminTotal.toFixed(1);
            document.getElementById('kpi-total-admin-max').innerText = adminMaxTotal;
        }

        // Admin Lưu điểm
        function saveKPIRatingByAdmin() {
            if (!isAdmin) return;
            const now = new Date();
            const dateStr = String(now.getDate()).padStart(2, '0') + '/' + 
                            String(now.getMonth() + 1).padStart(2, '0') + '/' + 
                            now.getFullYear();
            const timeStr = String(now.getHours()).padStart(2, '0') + ':' + 
                            String(now.getMinutes()).padStart(2, '0') + ':' + 
                            String(now.getSeconds()).padStart(2, '0');
            
            const fullTimestamp = `${timeStr} - ${dateStr}`;
            lastKpiTimestamp = fullTimestamp;

            saveKPIRatingStorage(fullTimestamp);
            alert(`Đã lưu kết quả đánh giá KPI cho tài khoản [${kpiTargetUser}] thành công lúc ${fullTimestamp}!`);
            renderKPITable();
        }

            saveKPIRatingStorage();
            alert("Đã lưu kết quả TỰ CHẤM điểm KPI thành công!");

        }

        // Cập nhật biểu đồ
        function updateCharts() {
            if (!statusChartInstance || !priorityChartInstance) return;

            const dataSource = (currentUser === ADMIN_USERNAME) ? getAllUsersTasks() : tasks;

            let todo = dataSource.filter(t => t.status === 'Chưa làm').length;
            let doing = dataSource.filter(t => t.status === 'Đang làm').length;
            let done = dataSource.filter(t => t.status === 'Hoàn thành').length;

            statusChartInstance.data.labels = [`Chưa làm (${todo})`, `Đang làm (${doing})`, `Hoàn thành (${done})`];
            statusChartInstance.data.datasets[0].data = [todo, doing, done];
            statusChartInstance.update();

            let high = dataSource.filter(t => t.priority === 'Cao').length;
            let normal = dataSource.filter(t => t.priority === 'Bình thường').length;
            let low = dataSource.filter(t => t.priority === 'Thấp').length;

            priorityChartInstance.data.datasets[0].data = [high, normal, low];
            priorityChartInstance.update();
        }

                scopeLabel.innerText = (currentUser === ADMIN_USERNAME) ? '(Thống kê tổng hợp toàn hệ thống)' : '(Thống kê cá nhân)';

            let todo = dataSource.filter(t => t.status === 'Chưa làm').length;

            });

            let high = dataSource.filter(t => t.priority === 'Cao').length;
            let normal = dataSource.filter(t => t.priority === 'Bình thường').length;
